## 一、问题分析

### 现状
- 系统只关注"孩子的记录"
- 缺少"家长自己的反思"
- 滋养报告里只有孩子的滋养时刻，没有家长的成长

### 理想状态
- 家长可以记录自己的反思和觉察
- 反思内容影响画像中"家长认知"的权重
- 滋养报告可以引用家长的反思

## 二、方案设计

### 2.1 数据库设计

```sql
-- 家长反思表（新增）
CREATE TABLE IF NOT EXISTS parent_reflections (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  child_id UUID REFERENCES children(id) ON DELETE SET NULL,
  
  content TEXT NOT NULL,                    -- 反思内容
  related_record_id UUID,                   -- 关联的陪伴记录（可选）
  related_opportunity_id UUID,               -- 关联的机会窗口（可选）
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_reflections_user ON parent_reflections(user_id);
CREATE INDEX IF NOT EXISTS idx_reflections_child ON parent_reflections(child_id);
```

## 四、前端组件

### 4.1 陪伴记录页面的反思区

```tsx
// src/app/(tab)/daily-care/page.tsx 新增组件

const REFLECTION_PROMPTS = [
  "今天记录完，我有什么新的觉察？",
  "在陪伴孩子的过程中，我有什么感受？",
  "上周的养育方式，有什么让我满意的？",
  "有什么我想尝试改变的地方？",
  "我对孩子的期待，和实际观察到的有什么差异？",
];

function ReflectionSection() {
  const [expanded, setExpanded] = useState(false);
  const [content, setContent] = useState('');
  const [saving, setSaving] = useState(false);
  const [saved, setSaved] = useState(false);
  const { currentChildId } = useChild();
  
  async function handleSave() {
    if (!content.trim()) return;
    
    setSaving(true);
    try {
      await reflectionsApi.create({
        childId: currentChildId,
        content: content.trim(),
      });
      setSaved(true);
      setContent('');
      setExpanded(false);
    } catch (e) {
      console.error('保存失败', e);
    } finally {
      setSaving(false);
    }
  }
  
  if (saved && !expanded) {
    return (
      <div className="mx-4 mt-4 bg-green-50 rounded-xl p-3">
        <p className="text-sm text-green-700">✅ 反思已记录 💭</p>
        <button 
          onClick={() => setExpanded(true)}
          className="text-xs text-green-600 mt-1"
        >
          再写一条
        </button>
      </div>
    );
  }
  
  return (
    <div className="mx-4 mt-4 bg-white rounded-xl overflow-hidden">
      <button
        onClick={() => setExpanded(!expanded)}
        className="w-full p-3 text-left text-sm text-gray-500 flex items-center justify-between"
      >
        <span>💭 家长反思</span>
        <span>{expanded ? '∧' : '∨'}</span>
      </button>
      
      {expanded && (
        <div className="p-3 pt-0 border-t">
          <p className="text-xs text-gray-400 mb-2">
            💡 {REFLECTION_PROMPTS[Math.floor(Math.random() * REFLECTION_PROMPTS.length)]}
          </p>
          <textarea
            value={content}
            onChange={(e) => setContent(e.target.value)}
            placeholder="写下你的反思和觉察..."
            rows={3}
            className="w-full p-3 border rounded-xl text-sm resize-none focus:outline-none focus:border-amber-400"
          />
          <div className="flex justify-end gap-2 mt-2">
            <button
              onClick={() => setExpanded(false)}
              className="px-4 py-2 text-sm text-gray-500"
            >
              收起
            </button>
            <button
              onClick={handleSave}
              disabled={!content.trim() || saving}
              className="px-4 py-2 bg-amber-500 text-white text-sm rounded-full disabled:opacity-50"
            >
              {saving ? '保存中...' : '保存'}
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
```

### 4.2 反思列表页

```tsx
// src/app/(tab)/profile/reflections/page.tsx

function ReflectionsPage() {
  const router = useRouter();
  const { currentChildId } = useChild();
  const [reflections, setReflections] = useState([]);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    loadReflections();
  }, []);
  
  async function loadReflections() {
    setLoading(true);
    try {
      const data = await reflectionsApi.list({ childId: currentChildId });
      setReflections(data.reflections);
    } catch (e) {
      console.error('加载失败', e);
    } finally {
      setLoading(false);
    }
  }
  
  if (loading) {
    return <div className="text-center py-8 text-gray-400">加载中...</div>;
  }
  
  return (
    <div className="min-h-screen bg-gradient-to-b from-amber-50 to-white pb-20">
      {/* Header */}
      <div className="sticky top-0 bg-white/90 backdrop-blur z-10 border-b">
        <div className="flex items-center gap-4 px-4 h-14">
          <button onClick={() => router.back()} className="text-gray-600">←</button>
```

### 4.3 API 封装

```typescript
// src/lib/api.ts 新增

export const reflectionsApi = {
  create: (data: { childId?: string; content: string; relatedRecordId?: string }) =>
    request('/api/reflections', {
      method: 'POST',
      body: JSON.stringify(data),
    }),
  
  list: (params: { childId?: string; limit?: number; offset?: number }) => {
    const qs = new URLSearchParams();
    if (params.childId) qs.set('childId', params.childId);
    if (params.limit) qs.set('limit', params.limit.toString());
    if (params.offset) qs.set('offset', params.offset.toString());
    const query = qs.toString();
    return request<{ reflections: ParentReflection[]; total: number }>(
      `/api/reflections${query ? '?' + query : ''}`
    );
  },
  
  update: (id: string, content: string) =>
    request(`/api/reflections/${id}`, {
      method: 'PATCH',
      body: JSON.stringify({ content }),
    }),
  
  delete: (id: string) =>
    request(`/api/reflections/${id}`, { method: 'DELETE' }),
};

interface ParentReflection {
  id: string;
  childId?: string;
  content: string;
  relatedRecordId?: string;
  createdAt: string;
}
```

## 五、与画像的关联

```typescript
// 反思影响家长认知权重
async function updateParentWeightFromReflection(userId: string, childId: string) {
  // 获取最近7天的反思数量
  const recentReflections = await query(
    `SELECT COUNT(*) as count FROM parent_reflections 
     WHERE user_id = $1 AND child_id = $2 AND created_at > NOW() - INTERVAL '7 days'`,
    [userId, childId]
  );
  
  const count = recentReflections[0]?.count || 0;
  
  // 反思越多，家长认知权重越高
  // 每个反思 +0.02，最多 +0.1
  const weightBonus = Math.min(count * 0.02, 0.1);
  
  // 更新画像权重（不超过上限 0.8）
  await query(
    `UPDATE child_profiles 
     SET parent_weight = LEAST(parent_weight + $1, 0.8) 
     WHERE child_id = $2`,
    [weightBonus, childId]
  );
}
```
