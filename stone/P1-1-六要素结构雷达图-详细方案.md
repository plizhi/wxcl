## 一、问题分析

### 现状
- AI分析基于"内在结构养育"的六要素框架
- 但用户无法直观看到孩子在六要素上的状态
- 缺乏可视化展示，只能看文字描述

### 理想状态
- 用雷达图展示孩子在六要素上的状态
- 支持时间维度对比（本周 vs 上周）
- 每个要素点开能看到具体事件支撑

## 三、评分算法

```typescript
// 六要素评分算法
function calculateElementScores(events: ProfileEvent[]): SixElementScores {
  const scores = {
    mind_spirit: { score: 0, eventCount: 0, evidence: [] },
    self_intention: { score: 0, eventCount: 0, evidence: [] },
    instinct_emotion: { score: 0, eventCount: 0, evidence: [] },
    internalized_object: { score: 0, eventCount: 0, evidence: [] },
    defense_mechanism: { score: 0, eventCount: 0, evidence: [] },
    inner_standards: { score: 0, eventCount: 0, evidence: [] },
  };
  
  for (const event of events) {
    // 根据事件类型和内容，计算各要素得分
    const elementScores = analyzeEvent(event);
    
    for (const [element, result] of Object.entries(elementScores)) {
      if (result.positive) {
        scores[element].score += result.value;
        scores[element].eventCount++;
        scores[element].evidence.push({
          eventId: event.id,
          fact: event.fact,
          interpretation: event.interpretation,
        });
      }
    }
  }
  
  // 归一化到 0-100
  for (const element of Object.keys(scores)) {
    const maxPossible = scores[element].eventCount * 20; // 每个事件最多20分
    scores[element].score = Math.min(100, (scores[element].score / maxPossible) * 100);
  }
  
  return scores;
}

// 事件分析
function analyzeEvent(event: ProfileEvent): ElementAnalysis {
  // 根据事件类型和内容，判断对各要素的影响
  const analysis = {
    mind_spirit: { positive: false, value: 0 },
    self_intention: { positive: false, value: 0 },
    instinct_emotion: { positive: false, value: 0 },
    internalized_object: { positive: false, value: 0 },
    defense_mechanism: { positive: false, value: 0 },
    inner_standards: { positive: false, value: 0 },
  };
  
  // 示例规则
  const content = event.fact.toLowerCase();
  
  // 心神：情绪觉察、表达身心状态
  if (content.includes('累了') || content.includes('困了') || 
      content.includes('不舒服') || content.includes('感知') ||
      content.includes('觉察')) {
    analysis.mind_spirit = { positive: true, value: 15 };
  }
  
  // 自我意向：表达自我、追求目标
  if (content.includes('我要') || content.includes('我想') ||
      content.includes('我的目标是') || content.includes('我想成为')) {
    analysis.self_intention = { positive: true, value: 15 };
  }
  
  // 内化客体：与重要他人的关系、安全感
  if (content.includes('妈妈') || content.includes('爸爸') ||
      content.includes('家人') || content.includes('安全感') ||
      content.includes('信任')) {
    analysis.internalized_object = { positive: true, value: 15 };
  }
  
  // 防御机制：压力应对方式（需要区分积极和消极）
  if (content.includes('逃避') || content.includes('压抑')) {
    analysis.defense_mechanism = { positive: false, value: -10 }; // 扣分
  }
  if (content.includes('沟通') || content.includes('表达')) {
    analysis.defense_mechanism = { positive: true, value: 15 };
  }
  
  // 内在准则：道德判断、责任意识
  if (content.includes('应该') || content.includes('不对') ||
      content.includes('责任') || content.includes('规则意识')) {
    analysis.inner_standards = { positive: true, value: 15 };
  }
  
  // 本能情感：情绪表达、健康需求
  if (content.includes('开心') || content.includes('生气') ||
      content.includes('难过') || content.includes('想要')) {
    analysis.instinct_emotion = { positive: true, value: 15 };
  }
  
  return analysis;
}
```

    impact: 'positive' | 'negative';

// src/components/profile/radar-chart.tsx

import { Radar } from 'recharts';  // 或使用 Chart.js

interface RadarChartProps {
  current: SixElementScores;
  previous?: SixElementScores;
  onElementClick: (element: string) => void;
}

function ProfileRadarChart({ current, previous, onElementClick }: RadarChartProps) {
  const data = SIX_ELEMENTS.elements.map(el => ({
    element: el.key,
    name: el.name,
    icon: el.icon,
    current: current[el.key]?.score || 0,
    previous: previous?.[el.key]?.score || 0,
  }));
  
  return (
    <div className="relative">
      {/* 雷达图 */}
      <ResponsiveContainer width="100%" height={300}>
        <RadarChart cx="50%" cy="50%" outerRadius="70%">
          <PolarGrid />
          <PolarAngleAxis dataKey="name" tickFormatter={(v) => {
            const el = SIX_ELEMENTS.elements.find(e => e.name === v);
            return el?.icon || v;
          }} />
          <PolarRadiusAxis angle={30} domain={[0, 100]} tickCount={5} />
          
          {/* 上周数据 */}
          {previous && (
            <Radar
              name="上周"
              dataKey="previous"
              stroke="#94a3b8"
              fill="#94a3b8"
              fillOpacity={0.1}
              strokeDasharray="5 5"
            />
          )}
          
          {/* 本周数据 */}
          <Radar
            name="本周"
            dataKey="current"
            stroke="#f59e0b"
            fill="#f59e0b"
            fillOpacity={0.3}
            strokeWidth={2}
          />
        </RadarChart>
      </ResponsiveContainer>
      
      {/* 图例 */}
      <div className="flex justify-center gap-6 mt-4">
        <div className="flex items-center gap-2">
          <div className="w-4 h-0.5 bg-amber-500 rounded" />
          <span className="text-sm text-gray-600">本周</span>
        </div>
        {previous && (
          <div className="flex items-center gap-2">
            <div className="w-4 h-0.5 bg-gray-400 rounded border-dashed" style={{borderStyle: 'dashed'}} />
            <span className="text-sm text-gray-600">上周</span>
          </div>
        )}
      </div>
      
      {/* 要素列表（可点击） */}
      <div className="mt-4 space-y-2">
        {data.map(item => (
          <button
            key={item.element}
            onClick={() => onElementClick(item.element)}
            className="w-full flex items-center justify-between p-2 hover:bg-gray-50 rounded-lg"
          >
            <div className="flex items-center gap-2">
              <span>{item.icon}</span>
              <span className="text-sm">{item.name}</span>
            </div>
            <div className="flex items-center gap-2">
              <div className="w-20 h-2 bg-gray-100 rounded-full overflow-hidden">
                <div 
                  className="h-full bg-amber-500 rounded-full"
                  style={{ width: `${item.current}%` }}
                />
              </div>
              <span className="text-sm font-medium w-10 text-right">{item.current}%</span>
              {item.current !== item.previous && (
                <span className={`text-xs ${item.current > item.previous ? 'text-green-500' : 'text-red-500'}`}>
                  {item.current > item.previous ? '↑' : '↓'}
                  {Math.abs(item.current - item.previous)}%
                </span>
              )}
            </div>
          </button>
        ))}
      </div>
    </div>
  );
}

// src/app/(tab)/profile/radar/[element]/page.tsx

function ElementDetailPage({ params }: { params: { element: string } }) {
  const router = useRouter();
  const { element } = params;
  const [detail, setDetail] = useState<ElementDetail | null>(null);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    loadDetail();
  }, [element]);
  
  async function loadDetail() {
    setLoading(true);
    try {
      const data = await profileApi.getElementDetail(element);
      setDetail(data);
    } catch (e) {
      console.error('加载失败', e);
    } finally {
      setLoading(false);
    }
  }
  
  if (loading) return <div>加载中...</div>;
  if (!detail) return <div>未找到数据</div>;
  
  const elementInfo = SIX_ELEMENTS.elements.find(e => e.key === element);
  
  return (
    <div className="min-h-screen bg-gradient-to-b from-amber-50 to-white pb-20">
      {/* Header */}
      <div className="sticky top-0 bg-white/90 backdrop-blur z-10 border-b">
        <div className="flex items-center gap-4 px-4 h-14">
          <button onClick={() => router.back()} className="text-gray-600">←</button>
          <h1 className="text-lg font-medium">
            {elementInfo?.icon} {elementInfo?.name}
          </h1>
        </div>
      </div>
      
      {/* 评分 */}
      <div className="mx-4 mt-4 bg-white rounded-2xl p-4 shadow-sm">
        <div className="text-sm text-gray-500 mb-2">综合评分</div>
        <div className="flex items-center gap-4">
          <div className="flex-1 h-4 bg-gray-100 rounded-full overflow-hidden">

          </div>
          <span className="text-lg font-medium text-amber-600">{detail.score}%</span>

      {/* 最近表现 */}
      <div className="mx-4 mt-4">
        <h3 className="text-sm font-medium text-gray-500 mb-2">最近表现</h3>
        <div className="space-y-3">

// src/app/(tab)/profile/page.tsx 新增 tab

function ProfilePage() {
  const [activeTab, setActiveTab] = useState<'overview' | 'radar'>('overview');
  const [radarData, setRadarData] = useState(null);
  
  // 切换到雷达图 tab
  function handleTabChange(tab: 'overview' | 'radar') {
    setActiveTab(tab);
    if (tab === 'radar') {
      loadRadarData();
    }
  }
  
  // 加载雷达图数据
  async function loadRadarData() {
    const data = await profileApi.getRadarData({
      childId: currentChildId,
      compareWithPrevious: true,
    });
    setRadarData(data);
  }
  
  // 要素点击，进入详情
  function handleElementClick(element: string) {
    router.push(`/profile/radar/${element}?childId=${currentChildId}`);
  }
  
  return (
    <div>
      {/* Tab 切换 */}
      <div className="flex border-b">
        <button 
          onClick={() => handleTabChange('overview')}
          className={`flex-1 py-3 text-sm ${activeTab === 'overview' ? 'border-b-2 border-amber-500 text-amber-600' : 'text-gray-500'}`}
        >
          画像概览
        </button>
        <button 
          onClick={() => handleTabChange('radar')}
          className={`flex-1 py-3 text-sm ${activeTab === 'radar' ? 'border-b-2 border-amber-500 text-amber-600' : 'text-gray-500'}`}
        >
          六要素结构
        </button>
      </div>
      
      {/* 内容 */}
      {activeTab === 'overview' && <ProfileOverview />}
      {activeTab === 'radar' && radarData && (
        <div className="p-4">
          <ProfileRadarChart
            current={radarData.current}
            previous={radarData.previous}
            onElementClick={handleElementClick}
          />
        </div>
      )}
    </div>
  );
}
