## 一、问题分析

### 现状
- 数据库设计了"家长认知（显性）"和"AI分析（隐性）"两套体系
- 但前端完全没有体现这个差异
- 用户只能看到修改后的结果，看不到系统内部的AI分析

### 理想状态
- 画像页面清晰展示"系统洞察"和"我的理解"的对比
- 用不同颜色/透明度区分两个版本
- 家长修改后，系统保持隐性分析不丢失
- 让家长感受到"AI在认真分析"但"我有最终决定权"

    challenges: string[],

  };
  
  // 动态权重
  parent_weight: number;  // 0.0-1.0，家长认知权重
}

## 三、API 设计

### 3.1 获取双版本数据

```typescript
// GET /api/profiles?childId=xxx&includeAI=true

Response: {
  profile: {
    // 家长版本（显性）
    parentVersion: {
      personality: { type, details },
      interests: string[],
      strengths: string[],
      challenges: string[],
      coreNeeds: string[],
      growthGoals: { enhancements, supports },
      lastModifiedAt: string,
      modifiedBy: 'parent' | 'ai',
    },
    
    // AI版本（隐性）
    aiVersion: {
      personality: { type, details, confidence },
      interests: string[],
      strengths: string[],
      challenges: string[],
      coreNeeds: string[],
      growthGoals: { enhancements, supports },
      reasoning: string,  // AI分析理由
      basedOnRecords: number,  // 基于多少条记录
    },
    
    // 权重
    parentWeight: number,  // 0.0-1.0
  }
}
```

// src/components/profile/profile-view.tsx

interface ProfileViewProps {
  profile: ChildProfile;
  aiVersion?: AIVersion;

function ProfileView({ profile, aiVersion, onEdit }: ProfileViewProps) {
  const [showAIDetails, setShowAIDetails] = useState(false);
  
  return (
    <div className="space-y-4">
      {/* 性格特质 */}
      <ProfileSection
        title="性格特质"
        icon="👤"
        parentValue={profile.personality}
        aiValue={aiVersion?.personality}

          </div>
        )}

            </div>

              <p className="text-xs text-gray-500 mt-1">{item.reasoning}</p>
            )}

        )}

      <ProfileSection
        title="优势强项"
        icon="✨"
        parentValue={profile.strengths}
        aiValue={aiVersion?.strengths}
        onEdit={onEdit}
        renderItem={(items) => (
          <div className="space-y-1">
            {items.map((s, i) => (
              <div key={i} className="flex items-center gap-2 text-sm text-gray-700">
                <span className="text-amber-500">✨</span> {s}
              </div>
            ))}
          </div>

          <div className="bg-blue-50 rounded-lg p-3">
            <div className="text-xs text-blue-600 mb-2">🤖 AI 从记录中发现</div>

// 通用区块组件
function ProfileSection({ title, icon, parentValue, aiValue, renderItem, renderAIItem }) {
  const [expanded, setExpanded] = useState(false);
  
  return (
    <div className="bg-white rounded-xl p-4 shadow-sm">
      <div className="flex items-center justify-between mb-2">
        <div className="flex items-center gap-2">
          <span>{icon}</span>
          <span className="font-medium text-gray-700">{title}</span>
        </div>
        <button onClick={() => setExpanded(!expanded)} className="text-xs text-amber-500">
          {expanded ? '收起' : '展开'}
        </button>
      </div>
      
      {/* 家长版本（始终显示） */}
      {parentValue && renderItem(parentValue)}
      
      {/* AI版本（可展开） */}
      {expanded && aiValue && (
        <div className="mt-3 pt-3 border-t border-gray-100">
          {renderAIItem(aiValue)}
        </div>
      )}
    </div>
  );
}

// src/components/profile/profile-edit.tsx

function ProfileEdit({ profile, aiVersion, onSave, onCancel }) {
  const [form, setForm] = useState({
    personality: profile.personality,
    interests: profile.interests || [],
    strengths: profile.strengths || [],
    challenges: profile.challenges || [],
    coreNeeds: profile.coreNeeds || [],
    growthGoals: profile.growthGoals || { enhancements: [], supports: [] },
  });
  
  // AI建议的增补（家长尚未确认）
  const aiSuggestions = {
    strengths: aiVersion?.strengths?.filter(
      s => !form.strengths.includes(s)
    ) || [],
    // ... 其他字段类似
  };
  
  return (
    <div className="space-y-4">
      {/* 性格特质 */}
      <div className="bg-white rounded-xl p-4">
        <div className="text-sm text-gray-500 mb-2">性格特质</div>
        <div className="text-sm text-gray-700 mb-2">
          当前：{form.personality.type === 'introvert' ? '内向型' : ...}
        </div>
        
        {/* AI建议 */}
        {aiVersion && (
          <div className="bg-blue-50 rounded-lg p-3 mb-2">
            <div className="text-xs text-blue-600 mb-1">💡 系统洞察建议（与您一致✓）</div>
            <div className="text-sm text-gray-700">
              {aiVersion.personality.type === form.personality.type ? '✓ 一致' : 
               `→ 建议改为：${aiVersion.personality.type}`}
            </div>
          </div>
        )}
        
        {/* 编辑选项 */}
        <div className="flex gap-2">
          {['introvert', 'extrovert', 'mixed'].map(type => (
            <button
              key={type}
              onClick={() => setForm(f => ({
                ...f,
                personality: { ...f.personality, type }
              }))}
              className={`px-4 py-2 rounded-full text-sm ${
                form.personality.type === type 
                  ? 'bg-amber-500 text-white' 
                  : 'bg-gray-100 text-gray-700'
              }`}
            >
              {type === 'introvert' ? '内向' : type === 'extrovert' ? '外向' : '混合'}
            </button>
          ))}
        </div>
      </div>
      
      {/* 优势强项（带AI建议） */}
      <div className="bg-white rounded-xl p-4">
        <div className="text-sm text-gray-500 mb-2">优势强项</div>
        
        {/* 当前值 */}
        <div className="flex flex-wrap gap-2 mb-3">
          {form.strengths.map((s, i) => (
            <span key={i} className="px-3 py-1 bg-amber-100 text-amber-700 rounded-full text-sm">
              {s}
            </span>
          ))}
        </div>
        
        {/* AI建议增补 */}
        {aiSuggestions.strengths.length > 0 && (
          <div className="bg-blue-50 rounded-lg p-3">
            <div className="text-xs text-blue-600 mb-2">💡 系统洞察新增建议</div>
            <div className="flex flex-wrap gap-2">
              {aiSuggestions.strengths.map((s, i) => (
                <button
                  key={i}
                  onClick={() => setForm(f => ({
                    ...f,
                    strengths: [...f.strengths, s]
                  }))}
                  className="px-3 py-1 border border-blue-300 text-blue-600 rounded-full text-xs hover:bg-blue-100"
                >
                  + {s}
                </button>
              ))}
            </div>
          </div>
        )}
      </div>
      
      {/* 操作按钮 */}
      <div className="flex gap-2">
        <button onClick={onCancel} className="flex-1 py-3 border rounded-full text-gray-500">
          取消
        </button>
        <button onClick={() => onSave(form)} className="flex-1 py-3 bg-amber-500 text-white rounded-full">
          保存
        </button>
      </div>
    </div>
  );
}

## 五、权重机制

```typescript
// 权重更新规则
const WEIGHT_RULES = {
  // 家长首次修改
  INITIAL_PARENT_EDIT: {
    condition: (profile) => profile.modifiedBy === 'ai',
    adjustment: 0.3,  // 初始家长权重设为0.3
  },
  
  // 家长确认AI分析
  PARENT_CONFIRMS_AI: {
    condition: (profile, edit) => edit.matchesAI === true,
    adjustment: 0.1,  // 权重+0.1
  },
  
  // 家长修改AI分析
  PARENT_MODIFIES_AI: {
    condition: (profile, edit) => edit.modifiedFromAI === true,
    adjustment: -0.05,  // 权重-0.05（小幅降低）
  },
  
  // 家长创建的内容
  PARENT_CREATES: {
    condition: (profile, edit) => edit.isNew === true,
    adjustment: 0.15,  // 权重+0.15
  },
};

// 权重边界
const WEIGHT_BOUNDARY = {
  min: 0.2,   // 最低权重（AI分析仍占主导）
  max: 0.8,   // 最高权重（家长认知占主导）
};
```
