## 一、问题分析

### 现状
- 目前只有全景报告页面有分享海报功能
- 使用 html2canvas 生成静态图片
- 海报类型单一

### 理想状态
- 支持多种海报模板：全景报告版、滋养时刻版、里程碑版
- 海报设计更精美，有品牌感
- 可以一键分享到社交媒体

## 三、前端组件

### 3.1 海报生成器

```tsx
// src/components/poster/poster-generator.tsx

import html2canvas from 'html2canvas';

interface PosterGeneratorProps {
  type: 'comprehensive_report' | 'nourishment' | 'milestone' | 'daily_record';
  data: PosterData;
  onGenerated: (dataUrl: string) => void;
}

async function generatePoster(
  type: string,
  data: PosterData,
  targetElement: HTMLElement
): Promise<string> {
  // 隐藏不需要的元素
  const originalDisplay = targetElement.style.display;
  targetElement.style.display = 'block';
  
  const canvas = await html2canvas(targetElement, {
    backgroundColor: BRAND_COLORS.background,
    scale: 2,  // 2倍清晰度
    useCORS: true,
    logging: false,
    width: POSTER_SIZE.width / 2,  // 缩放
    height: POSTER_SIZE.height / 2,
  });
  
  // 恢复显示
  targetElement.style.display = originalDisplay;
  
  return canvas.toDataURL('image/png');
}
```

## 四、分享功能

```tsx
// 新增海报模板选择
function ShareModal() {
  const [selectedTemplate, setSelectedTemplate] = useState('comprehensive_report');
  const { toast } = useToast();
  
  const templates = [
    { type: 'comprehensive_report', name: '全景报告', icon: '📊' },
    { type: 'nourishment', name: '滋养时刻', icon: '💧' },
    { type: 'milestone', name: '成长里程碑', icon: '🏆' },
  ];
  
  async function handleShare() {
    try {
      const posterDataUrl = await generatePoster(
        selectedTemplate, 
        getPosterData(selectedTemplate),
        posterRef.current
      );
      
      const link = document.createElement('a');
      link.download = `望杏林${templates.find(t => t.type === selectedTemplate)?.name}_${new Date().toISOString().split('T')[0]}.png`;
      link.href = posterDataUrl;
      link.click();
      
      toast('海报已保存', 'success');
    } catch (e) {
      toast('生成失败', 'error');
    }
  }
  
  return (
    <div className="p-4">
      <h3 className="text-lg font-medium mb-4">选择海报模板</h3>
      
      {/* 模板选择 */}
      <div className="grid grid-cols-3 gap-2 mb-4">
        {templates.map(t => (
          <button
            key={t.type}
            onClick={() => setSelectedTemplate(t.type)}
            className={`p-3 rounded-xl text-center ${
              selectedTemplate === t.type 
                ? 'bg-amber-100 ring-2 ring-amber-500' 
                : 'bg-gray-50'
            }`}
          >
            <div className="text-2xl mb-1">{t.icon}</div>
            <div className="text-xs">{t.name}</div>
          </button>
        ))}
      </div>
      
      {/* 预览 */}
      <div className="mb-4">
        <PosterPreview type={selectedTemplate} data={getPosterData(selectedTemplate)} />
      </div>
      
      {/* 分享按钮 */}
      <button
        onClick={handleShare}
        className="w-full py-3 bg-gradient-to-r from-amber-500 to-orange-500 text-white rounded-full font-medium"
      >
        保存海报
      </button>
    </div>
  );
}
```
