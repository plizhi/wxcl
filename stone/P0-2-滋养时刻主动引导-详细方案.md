> 创建时间：2026-07-29
> 状态：待评审

## 一、问题分析

### 现状
- 系统设计了「从记录中提取滋养时刻」的功能（`/api/nourishment/extract`）
- 但用户不知道这个功能存在，也不知道什么时候该用它
- 滋养时刻页面只有一个手动添加入口，没有引导用户从陪伴记录中提取

### 理想状态
- 陪伴记录提交后，系统自动检测是否有滋养时刻，并主动提示用户
- 用户可以一键确认保存，也可以修改后保存
- 在滋养时刻不足时，适时引导用户去提取历史记录

## 三、API 设计

### 3.1 改造 extract 接口

```typescript
// POST /api/nourishment/extract

// 现有逻辑：批量提取，默默保存
// 改造后：支持"预览模式"和"保存模式"

Request: {
  childId?: string;
  limit?: number;           // 默认 10
  mode: 'preview' | 'save'; // 新增：预览或保存
}

Response (preview 模式): {
  extractions: Array<{
    fact: string;           // 滋养事实
    feeling: string;        // 建议的感受
    sourceRecordId: string; // 来源记录ID
    sourceRecordPreview: string; // 来源记录摘要（用于展示）
  }>;
  totalAvailable: number;  // 可提取的总数
}

Response (save 模式): {
  extractions: Array<...>;  // 同上
  savedCount: number;       // 实际保存数量
  totalAvailable: number;   // 剩余可提取
}
```

### 3.2 新增 API 端点

```typescript
// POST /api/nourishment/extract-single
// 提取单条记录中的滋养时刻（陪伴记录提交后调用）

Request: {
  recordId: string;
  fact: string;       // 事实（AI判断后可为空）
  feeling?: string;   // 感受（可选）
  save: boolean;       // true=直接保存，false=仅预览
}

Response: {
  moment?: NourishmentMoment;
  extracted: boolean;
  fact?: string;      // AI提取的事实
  feeling?: string;   // AI建议的感受
}
```

### 4.4 提取确认列表页

```tsx
// 新增页面：/src/app/(tab)/nourishment/extract/page.tsx

function NourishmentExtractPage() {
  const router = useRouter();
  const { currentChildId } = useChild();
  const [extractions, setExtractions] = useState<Extraction[]>([]);
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [loading, setLoading] = useState(false);
  
  async function handleExtract() {
    setLoading(true);
    try {
      const result = await nourishmentApi.extract({
        childId: currentChildId,
        mode: 'preview',
        limit: 20,
      });
      setExtractions(result.extractions);
      // 默认全选
      setSelected(new Set(result.extractions.map((_: any, i: number) => i.toString())));
    } catch (e) {
      console.error('提取失败', e);
    } finally {
      setLoading(false);
    }
  }
  
  function toggleSelect(index: string) {
    const newSelected = new Set(selected);
    if (newSelected.has(index)) {
      newSelected.delete(index);
    } else {
      newSelected.add(index);
    }
    setSelected(newSelected);
  }
  
  async function handleSaveSelected() {
    const toSave = extractions.filter((_: any, i: number) => selected.has(i.toString()));
    for (const item of toSave) {
      await nourishmentApi.addMoment({
        childId: currentChildId,
        fact: item.fact,
        feeling: item.feeling,
        source: 'extracted',
        extractedFromRecordId: item.sourceRecordId,
      });
    }
    toast(`已保存 ${toSave.length} 条滋养时刻 💧`, 'success');
    router.back();
  }
  
  if (extractions.length === 0 && !loading) {
    return (
      <div className="min-h-screen bg-gradient-to-b from-purple-50 to-white flex flex-col items-center justify-center">
        <p className="text-gray-500 mb-4">暂无待提取的滋养时刻</p>
        <button onClick={handleExtract} className="px-6 py-2 bg-purple-500 text-white rounded-full">
          刷新检测
        </button>
      </div>
    );
  }
  
  return (
    <div className="min-h-screen bg-gradient-to-b from-purple-50 to-white pb-20">
      {/* Header */}
      <div className="sticky top-0 bg-white/90 backdrop-blur z-10 border-b">
        <div className="flex items-center justify-between px-4 h-14">
          <button onClick={() => router.back()} className="text-gray-600">←</button>
          <h1 className="text-lg font-medium">提取滋养时刻</h1>
          <span className="text-sm text-purple-500">{selected.size} 已选</span>
        </div>
      </div>
      
      {/* 列表 */}
      <div className="p-4 space-y-3">
        {extractions.map((item, i) => (
          <div 
            key={i} 
            className={`p-4 bg-white rounded-xl cursor-pointer transition-all ${
              selected.has(i.toString()) ? 'ring-2 ring-purple-400' : ''
            }`}
            onClick={() => toggleSelect(i.toString())}
          >
            <div className="flex items-start gap-3">
              <div className={`w-5 h-5 rounded border-2 flex items-center justify-center mt-0.5 ${
                selected.has(i.toString()) ? 'bg-purple-500 border-purple-500' : 'border-gray-300'
              }`}>
                {selected.has(i.toString()) && <span className="text-white text-xs">✓</span>}
              </div>
              <div className="flex-1">
                <p className="text-sm text-gray-700">{item.fact}</p>
                <p className="text-xs text-purple-600 mt-1">💧 {item.feeling}</p>
                <p className="text-xs text-gray-400 mt-1">来源：{item.sourceRecordPreview}</p>
              </div>
            </div>
          </div>
        ))}
      </div>
      
      {/* 底部操作栏 */}
      <div className="fixed bottom-0 left-0 right-0 bg-white border-t p-4">
        <button 
          onClick={handleSaveSelected} 
          disabled={selected.size === 0}
          className="w-full py-3 bg-purple-500 text-white rounded-full disabled:opacity-50"
        >
          保存所选 ({selected.size})
        </button>
      </div>
    </div>
  );
}
```

## 五、时机策略

```typescript
// 引导强度分级
const GUIDANCE_STRATEGY = {
  // 时机1：陪伴记录提交后
  AFTER_RECORD: {
    triggerCondition: (record) => true, // 每次都检测
    minInterval: 0, // 不限制间隔
    maxPerWeek: Infinity, // 不限制次数
  },
  
  // 时机2：进入滋养页面
  ON_NOURISHMENT_PAGE: {
    triggerCondition: (totalMoments) => totalMoments < 3,
    minInterval: 7 * 24 * 60 * 60 * 1000, // 7天只显示一次
    maxPerWeek: 1,
  },
  
  // 时机3：生成报告前
  BEFORE_REPORT: {
    triggerCondition: (totalMoments) => totalMoments < 5,
    minInterval: 0,
    maxPerWeek: Infinity,
  },
};
```
